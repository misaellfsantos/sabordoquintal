// ================= PERFIL DO USUÁRIO ================= // CARREGAR DADOS function pegarDadosLogin() { const nome = localStorage.getItem("nomeUsuario") || "Usuário"; const email = localStorage.getItem("emailUsuario") || "Não informado"; const telefone = localStorage.getItem("telefoneUsuario") || "Não informado"; const foto = localStorage.getItem("usuarioFoto"); const perfilNome = document.getElementById("perfilNome"); const perfilNomeDado = document.getElementById("perfilNomeDado"); const perfilEmail = document.getElementById("perfilEmail"); const perfilTelefone = document.getElementById("perfilTelefone"); const fotoPerfil = document.getElementById("fotoPerfil"); if (perfilNome) perfilNome.textContent = nome; if (perfilNomeDado) perfilNomeDado.textContent = nome; if (perfilEmail) perfilEmail.textContent = email; if (perfilTelefone) perfilTelefone.textContent = telefone; if (foto && fotoPerfil) { fotoPerfil.src = foto; } } // ABRIR PERFIL function abrirPerfil() { const usuarioLogado = localStorage.getItem("usuarioLogado"); if (usuarioLogado !== "true") { alert("Faça login para acessar o perfil."); window.location.href = "/login"; return; } pegarDadosLogin(); const telaPerfil = document.getElementById("telaPerfil"); if (telaPerfil) { telaPerfil.classList.add("ativa"); document.body.style.overflow = "hidden"; } else { console.error("Elemento telaPerfil não encontrado."); } } // FECHAR PERFIL function fecharPerfil() { const telaPerfil = document.getElementById("telaPerfil"); if (telaPerfil) { telaPerfil.classList.remove("ativa"); } document.body.style.overflow = ""; } // ALTERAR FOTO function configurarFotoPerfil() { const inputFoto = document.getElementById("inputFoto"); if (!inputFoto) return; inputFoto.addEventListener("change", function () { const arquivo = this.files[0]; if (!arquivo) return; if (!arquivo.type.startsWith("image/")) { alert("Selecione uma imagem válida."); return; } const leitor = new FileReader(); leitor.onload = function (evento) { const foto = evento.target.result; document.getElementById("fotoPerfil").src = foto; localStorage.setItem("usuarioFoto", foto); }; leitor.readAsDataURL(arquivo); }); } // ABRIR MINHAS RESERVAS async function abrirReservas() { const telaReservas = document.getElementById("telaReservas"); const lista = document.getElementById("listaReservas"); if (!telaReservas || !lista) { alert("Área de reservas não encontrada."); return; } telaReservas.classList.add("ativa"); lista.textContent = "Carregando reservas..."; const telefone = localStorage.getItem("telefoneUsuario"); if (!telefone) { lista.textContent = "Telefone não encontrado."; return; } try { const resposta = await fetch( "/api/reservas/minhas?telefone=" + encodeURIComponent(telefone) ); if (!resposta.ok) { throw new Error("Erro ao buscar reservas."); } const reservas = await resposta.json(); mostrarReservas(reservas); } catch (erro) { console.error(erro); lista.textContent = "Não foi possível carregar suas reservas."; } } // MOSTRAR RESERVAS function mostrarReservas(reservas) { const lista = document.getElementById("listaReservas"); if (!lista) return; lista.innerHTML = ""; if (!reservas || reservas.length === 0) { lista.textContent = "Você ainda não possui reservas."; return; } reservas.forEach(reserva => { const card = document.createElement("div"); card.className = "reserva-item"; const data = reserva.dataReserva ? reserva.dataReserva.split("-").reverse().join("/") : "Não informada"; const campos = [ ["Data", data], ["Horário", reserva.horario || "Não informado"], ["Pessoas", reserva.quantidadePessoas || "Não informado"], ["Ambiente", reserva.ambiente || "Não informado"], ["Mesa", reserva.mesa || "Não informada"], ["Status", reserva.status || "PENDENTE"] ]; const titulo = document.createElement("h3"); titulo.textContent = "Minha reserva"; card.appendChild(titulo); campos.forEach(([nome, valor]) => { const p = document.createElement("p"); const strong = document.createElement("strong"); strong.textContent = nome + ": "; p.appendChild(strong); p.appendChild(document.createTextNode(valor)); card.appendChild(p); }); lista.appendChild(card); }); } // FECHAR RESERVAS function fecharReservas() { const telaReservas = document.getElementById("telaReservas"); if (telaReservas) { telaReservas.classList.remove("ativa"); } } // SAIR DA CONTA function sairDaConta() { if (!confirm("Deseja realmente sair da sua conta?")) { return; } localStorage.removeItem("usuarioLogado"); fecharPerfil(); window.location.href = "/login"; } // INICIALIZAÇÃO document.addEventListener("DOMContentLoaded", function () { configurarFotoPerfil(); pegarDadosLogin(); document.addEventListener("keydown", function (evento) { if (evento.key === "Escape") { fecharReservas(); fecharPerfil(); } }); });
